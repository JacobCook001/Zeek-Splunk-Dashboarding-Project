  <!-- ── ROW 2: External RDP inbound ── -->
  <row>
    <panel>
      <!-- Inbound RDP from non-RFC1918 IPs is almost never legitimate in a
           well-managed network. VPNs should terminate before RDP is reached. T1021.001 -->
      <title>RDP from External IP (Non-RFC1918 Source) (T1021.001)</title>
      <table>
        <search>
          <query>index=zeek source="*conn.log" $src_ip_include$ $src_ip_exclude$ id.resp_p=3389
| where NOT match(id.orig_h, "^(10\.|172\.(1[6-9]|2[0-9]|3[01])\.|192\.168\.)")
| stats count AS connections, dc(id.resp_h) AS unique_targets, values(id.resp_h) AS targets,
        values(conn_state) AS states, min(_time) AS first_seen, max(_time) AS last_seen
    by id.orig_h
| eval first_seen=strftime(first_seen, "%Y-%m-%dT%H:%M:%S")
| eval last_seen=strftime(last_seen, "%Y-%m-%dT%H:%M:%S")
| sort - connections</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
        </search>

        </drilldown>
      </table>
    </panel>
    <panel>
      <!-- Internal → internal RDP is lateral movement when the source is a workstation
           rather than a known jump server / bastion. T1021.001 -->
      <title>RDP Lateral Movement (Internal → Internal) (T1021.001)</title>
      <table>
        <search>
          <query>index=zeek source="*conn.log" $src_ip_include$ $src_ip_exclude$ id.resp_p=3389
| where match(id.orig_h, "^(10\.|172\.(1[6-9]|2[0-9]|3[01])\.|192\.168\.)")
  AND match(id.resp_h, "^(10\.|172\.(1[6-9]|2[0-9]|3[01])\.|192\.168\.)")
| stats count AS connections, dc(id.resp_h) AS unique_targets, values(id.resp_h) AS targets,
        values(conn_state) AS states
    by id.orig_h
| sort - connections</query>

  <!-- ── ROW 3: Brute force ── -->
  <row>
    <panel>
      <!-- Brute force signature: many short-duration connections from same src to same target.
           Short duration + failed conn_state + high count = automated credential spray. T1110 -->
      <title>RDP Brute Force Candidates - Many Short Connections (&gt;10) (T1110)</title>
      <table>
        <search>
          <query>index=zeek source="*conn.log" $src_ip_include$ $src_ip_exclude$ id.resp_p=3389
| where duration < 10 OR isnull(duration)
| stats count AS attempts, dc(id.resp_h) AS unique_targets, values(conn_state) AS states
    by id.orig_h
| where attempts > 10
| sort - attempts</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
        </search>

    by id.orig_h

  <!-- ── ROW 4: Session duration analysis ── -->
  <row>
    <panel>
      <!-- Very short RDP sessions (< 30s) = failed auth or automated probe.
           Very long sessions (> 2hr) = interactive attacker activity. Both ends are worth examining. -->
      <title>RDP Session Duration Outliers (All Sessions)</title>
      <table>
        <search>
          <query>index=zeek source="*conn.log" $src_ip_include$ $src_ip_exclude$ id.resp_p=3389
| where isnotnull(duration)
| eval duration_sec=round(duration, 1)
| eval session_type=case(
    duration < 5,    "very short (&lt;5s) — probe/fail",
    duration < 30,   "short (&lt;30s) — likely failed auth",
    duration > 7200, "long (&gt;2hr) — interactive session",
    true(),          "normal")
| stats count AS sessions, avg(duration_sec) AS avg_sec by id.orig_h, id.resp_h, session_type
| eval avg_sec=round(avg_sec, 1)
| sort session_type, - sessions</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
        </search>
        <option name="count">20</option>
