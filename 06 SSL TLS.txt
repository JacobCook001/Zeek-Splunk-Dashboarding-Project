  <!-- ── ROW 2: Deprecated versions & weak ciphers ── -->
  <row>
    <panel>
      <!-- SSLv2/v3 and TLS 1.0/1.1 are deprecated and cryptographically broken.
           Any usage in a modern network warrants investigation. T1573 -->
      <title>Deprecated TLS Versions (SSLv2/SSLv3/TLSv10/TLSv11)</title>
      <table>
        <search>
          <query>index=zeek source="*ssl.log" $src_ip_include$ $src_ip_exclude$
| search version=SSLv2 OR version=SSLv3 OR version=TLSv10 OR version=TLSv11
| stats count AS sessions, dc(id.resp_h) AS unique_servers, values(server_name) AS snis by id.orig_h, version
| sort - sessions</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
        </search>

  <!-- ── ROW 3: Certificate issues ── -->
  <row>
    <panel>
      <!-- Self-signed: issuer == subject. Legitimate CAs never appear as their own issuer
           in a leaf cert. Common in C2 frameworks, malware, and attacker infra. -->
      <title>Self-Signed Certificates (Issuer = Subject)</title>
      <table>
        <search>
          <query>index=zeek source="*ssl.log" $src_ip_include$ $src_ip_exclude$
| where subject=issuer AND subject!="" AND subject!="-"
| stats count AS sessions, dc(id.orig_h) AS unique_clients by id.resp_h, id.resp_p, subject, server_name
| sort - sessions</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
        </search>
        <option name="count">20</option>

  <!-- ── ROW 4: JA3/JA3S fingerprinting ── -->
  <row>
    <panel>
      <!-- JA3 hashes the TLS ClientHello parameters (version, cipher list, extensions,
           elliptic curves). Known C2 frameworks have documented JA3 signatures.
           Requires Zeek ja3 package. -->
      <title>JA3 Client Fingerprints - Top (C2 Client Detection)</title>
      <table>
        <search>
          <query>index=zeek source="*ssl.log" $src_ip_include$ $src_ip_exclude$
| where ja3!="" AND ja3!="-"
| stats count AS sessions, dc(id.orig_h) AS unique_clients, dc(id.resp_h) AS unique_servers, values(server_name) AS snis by ja3
| sort - sessions</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
        </search>
        <option name="count">20</option>
        <option name="drilldown">cell</option>
        <option name="refresh.display">progressbar</option>
        <option name="wrap">true</option>
        <drilldown target="_blank">
          <link>search?q=index%3Dzeek+source%3D"*ssl.log"+ja3%3D"$row.ja3|u$"</link>
        </drilldown>
      </table>
    </panel>
    <panel>
      <!-- JA3S fingerprints the ServerHello. Combining JA3+JA3S can attribute traffic
           to a specific C2 framework and server implementation. -->
      <title>JA3S Server Fingerprints - Top</title>
      <table>
        <search>
          <query>index=zeek source="*ssl.log" $src_ip_include$ $src_ip_exclude$
| where ja3s!="" AND ja3s!="-"
| stats count AS sessions, dc(id.orig_h) AS unique_clients, dc(id.resp_h) AS unique_servers by ja3s
| sort - sessions</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
        </search>
        <option name="count">20</option>
        <option name="drilldown">cell</option>

      </table>
    </panel>
  </row>

        <search>

| where unique_ja3 > 3
| sort - unique_ja3</query>

      <!-- Modern browsers and legit apps always send SNI in TLS 1.2+.

      <table>
        <search>
          <query>index=zeek source="*ssl.log" $src_ip_include$ $src_ip_exclude$
| where (server_name="" OR server_name="-" OR isnull(server_name)) AND established=true
| stats count AS sessions, dc(id.resp_h) AS unique_servers, values(id.resp_h) AS servers, values(version) AS versions by id.orig_h
| sort - sessions</query>
