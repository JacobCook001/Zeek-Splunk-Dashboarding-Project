  <!-- DATA MODEL                                                          -->
  <!-- All "outbound" is defined as:                                       -->
  <!--   id.orig_h = RFC1918 internal host                                -->
  <!--   id.resp_h = non-RFC1918 external IP                              -->
  <!-- orig_bytes = bytes sent BY the internal host (the exfil direction) -->
  <!-- resp_bytes = bytes received FROM external (downloads, C2 replies)  -->
  <!--                                                                     -->
  <!-- RFC1918 ranges used throughout:                                     -->
  <!--   10.0.0.0/8      ^10\.                                            -->
  <!--   172.16-31.0/12  ^172\.(1[6-9]|2[0-9]|3[01])\.                  -->
  <!--   192.168.0.0/16  ^192\.168\.                                      -->
  <!--                                                                     -->
  <!-- Sources used:                                                       -->
  <!--   conn.log  — all IP/TCP/UDP flows, orig_bytes / resp_bytes        -->
  <!--   http.log  — request_body_len (POST/PUT uploads)                  -->
  <!--   ftp.log   — STOR/RETR, file_size                                 -->
  <!--   dns.log   — query length, volume (tunnelling)                    -->
  <!--   smtp.log  — outbound email + attachment fuids                    -->
  <!--   ssl.log   — encrypted channel byte volumes                       -->
  <!-- ═══════════════════════════════════════════════════════════════════ -->

      AND orig_bytes &gt; 0
| stats sum(orig_bytes) AS total_sent,
        sum(resp_bytes) AS total_recv,
        count AS connections,

| table id.orig_h, id.resp_h, sent_mb, recv_mb, out_in_ratio,
        connections, unique_ports, services, first_seen, last_seen

      AND (method="POST" OR method="PUT" OR method="PATCH")
      AND request_body_len &gt; 0
| eval req_kb = round(request_body_len/1024, 2)
| stats sum(request_body_len) AS total_upload,

        values(uri) AS uris,

| eval total_mb = round(total_upload/1048576, 2)
| where total_mb &gt; 0.01
| table id.orig_h, id.resp_h, id.resp_p, host, method,
        total_mb, requests, uris, user_agents, status_codes
| rename id.orig_h AS "Source", id.resp_h AS "Destination",

    (qtype_name=A OR qtype_name=AAAA OR qtype_name=TXT OR qtype_name=NULL OR qtype_name=MX)

| eval base_domain = replace(query, "^(?:[^.]+\.)+?([^.]+\.[^.]+)$", "\1")

    max_query_len &gt; 100 AND unique_subdomains &gt; 20,       "HIGH — long+random labels",
    max_query_len &gt; 50  AND queries &gt; 50,                 "MED  — long labels + volume",
    unique_subdomains &gt; 30,                                  "MED  — many subdomains",
