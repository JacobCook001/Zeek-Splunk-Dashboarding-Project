      <table>
        <title>SMB Share information</title>
        <search>
          <query>index=zeek source="*smb_files.log"
| rename id.orig_h as src_ip, id.resp_h as dest_ip
| stats
    count as total_actions,
    dc(name) as unique_files,
    dc(path) as unique_paths,
    dc(dest_ip) as unique_dest_hosts,
    values(action) as actions_performed,
    values(path) as paths_accessed,
    earliest(_time) as first_seen,
    latest(_time) as last_seen
    by src_ip
| eval duration_mins=round((last_seen-first_seen)/60, 1)
| convert ctime(first_seen) ctime(last_seen)
| sort -total_actions
| head 10
| rename
    src_ip as "Source IP",
    total_actions as "Total Actions",
    unique_files as "Unique Files",
    unique_paths as "Unique Paths",
    unique_dest_hosts as "Unique Dest Hosts",
    actions_performed as "Actions Performed",
    paths_accessed as "Paths Accessed",
    first_seen as "First Seen",
    last_seen as "Last Seen",
    duration_mins as "Duration (mins)"</query>
          <earliest>-4h@m</earliest>
          <latest>now</latest>
        </search>
        <option name="drilldown">none</option>
        <option name="refresh.display">progressbar</option>
