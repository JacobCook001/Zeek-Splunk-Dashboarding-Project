  <!-- ── ROW 2: Kerberoasting & AS-REP Roasting ── -->
  <row>
    <panel>
      <!-- T1558.003 — Kerberoasting
           Attackers request TGS tickets for service accounts using RC4-HMAC (etype 23)
           to crack offline. AES-only environments should have zero RC4 TGS tickets. -->
      <title>Kerberoasting Candidates - TGS with RC4-HMAC Cipher (T1558.003)</title>
      <table>
        <search>
          <query>index=zeek source="*kerberos.log" $src_ip_include$ $src_ip_exclude$
| search request_type=TGS cipher=rc4-hmac
| where match(service, "\/")
| stats count, values(uid) AS uids by id.orig_h, id.resp_h, client, service, cipher
| sort - count</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
        </search>
        <option name="count">20</option>
        <option name="drilldown">cell</option>
        <option name="refresh.display">progressbar</option>

        <drilldown target="_blank">
          <link>search?q=index%3Dzeek+source%3D"*kerberos.log"+id.orig_h%3D"$row.id.orig_h|u$"+request_type%3DTGS+cipher%3Drc4-hmac</link>
        </drilldown>
      </table>
    </panel>
    <panel>
      <!-- T1558.004 — AS-REP Roasting
           Accounts with "Do not require Kerberos preauthentication" return crackable
           AS-REP blobs. Network signature: rapid AS-REQ for many distinct principals
           from one source, often with short inter-request gaps. -->
      <title>AS-REP Roasting Candidates - Rapid AS-REQ Enumeration (T1558.004)</title>
      <table>
        <search>
          <query>index=zeek source="*kerberos.log" $src_ip_include$ $src_ip_exclude$
| search request_type=AS
| stats count AS as_requests, dc(client) AS unique_clients, values(client) AS clients, dc(service) AS unique_services by id.orig_h, id.resp_h
| where unique_clients > 3
| sort - as_requests</query>
          <earliest>$time_range.earliest$</earliest>

  <!-- ── ROW 5: Golden / Silver ticket indicators ── -->
  <row>
    <panel>
      <!-- T1558.001/002 — Golden/Silver Ticket
           Forged tickets may have anomalous lifetimes (till far in future or epoch 0).
           A TGS request arriving with no prior TGT in the window for the same client
           is also suspicious (attacker injected a golden ticket locally). -->
      <title>Anomalous Ticket Lifetimes - Golden/Silver Ticket Indicator (T1558.001/002)</title>
      <table>
        <search>
          <query>index=zeek source="*kerberos.log" $src_ip_include$ $src_ip_exclude$
| where isnotnull(till) AND till!=""
| eval till_epoch=strptime(till, "%Y-%m-%dT%H:%M:%S")
| eval till_days=round((till_epoch - now()) / 86400, 1)
| where till_days > 30 OR till_days < 0
| stats count by id.orig_h, id.resp_h, client, service, request_type, cipher, till, till_days, uid
| sort - till_days</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
        </search>
        <option name="count">20</option>
        <option name="drilldown">cell</option>
        <option name="refresh.display">progressbar</option>
        <option name="wrap">true</option>
        <drilldown target="_blank">
          <link>search?q=index%3Dzeek+source%3D"*kerberos.log"+uid%3D"$row.uid|u$"</link>
        </drilldown>
      </table>
    </panel>
    <panel>
      <!-- Sources making TGS requests with no corresponding AS-REQ in the window.
           Normal flow: AS-REQ → AS-REP → TGS-REQ → TGS-REP.
           TGS with no AS-REQ suggests a pre-existing (possibly forged) ticket. -->
      <title>TGS Without Prior AS-REQ (Same Source, Same Window)</title>
      <table>
        <search>
          <query>index=zeek source="*kerberos.log" $src_ip_include$ $src_ip_exclude$
| stats dc(request_type) AS type_count, values(request_type) AS types, count by id.orig_h, id.resp_h, client
| where type_count=1 AND types="TGS"
| sort - count</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>

  <row>
    <panel>
      <!-- T1550.002 — Pass-the-Hash / NTLM Relay
           A relay attack has one source authenticating to many distinct targets in a
           short window using the same username. Normal machines authenticate to a
           handful of DCs; > N unique targets from one host is suspicious. -->
      <title>NTLM Relay Candidates - One Source to Many Targets (T1550.002)</title>
      <table>
        <search>
          <query>index=zeek source="*ntlm.log" $src_ip_include$ $src_ip_exclude$
| stats dc(id.resp_h) AS unique_targets, dc(username) AS unique_users, count AS events, values(username) AS usernames, values(id.resp_h) AS targets by id.orig_h
| where unique_targets > 5
| sort - unique_targets</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
        </search>

      </table>
    </panel>
    <panel>
      <!-- Hosts appearing in ntlm.log but NOT in kerberos.log for the same window.
           In a Kerberos-enabled AD environment, workstations should use Kerberos.
           Pure NTLM from a host suggests: non-domain-joined attacker, PTH, or
           deliberate downgrade forcing NTLM negotiation. -->
      <title>Hosts Using NTLM Only (No Kerberos) - Potential PTH/Downgrade</title>
      <table>
        <search>
          <query>index=zeek (source="*ntlm.log" OR source="*kerberos.log") $src_ip_include$ $src_ip_exclude$
| eval proto=if(match(source, "ntlm"), "ntlm", "kerberos")
| stats dc(proto) AS proto_count, values(proto) AS protocols, count by id.orig_h
| where proto_count=1 AND protocols="ntlm"
| sort - count</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
