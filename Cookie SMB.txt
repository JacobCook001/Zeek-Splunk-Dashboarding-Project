      <!-- ═══════════════════════════════════════════════════════════════ -->
      <!-- FIX: The original query used | search path="*ADMIN$*" etc.    -->
      <!-- Splunk treats $ as a token delimiter in search strings, so     -->
      <!-- ADMIN$, C$, IPC$ were silently consumed as empty tokens and   -->
      <!-- the filter matched nothing. Fixed by moving share detection    -->
      <!-- into a | where clause using match() with regex, which is       -->
      <!-- evaluated after tokenisation and is immune to this issue.      -->
      <!-- ═══════════════════════════════════════════════════════════════ -->
      <title>SMB Admin Share Access (C$, ADMIN$, IPC$, SYSVOL, NETLOGON)</title>
      <table>
        <search>
          <query>index=zeek source="*smb_mapping.log" $src_ip_include$ $src_ip_exclude$
| where match(path, "(?i)(ADMIN\$|C\$|IPC\$|SYSVOL|NETLOGON)")
| stats count by id.orig_h, id.resp_h, path, share_type, uid
| sort - count</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
