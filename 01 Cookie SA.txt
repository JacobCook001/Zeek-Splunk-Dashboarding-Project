      <table>
        <search base="Base_Search">
          <query>| search $src_ip_include$ $src_ip_exclude$ (id.resp_p=443 OR id.resp_p=80 OR id.resp_p=8080)
| transaction id.orig_h, id.resp_h maxpause=15m
| eval interval = duration
| stats avg(interval) as avg_interval_sec, count as sessions by id.orig_h, id.resp_h
| eval avg_interval_sec = round(avg_interval_sec, 2)
| where sessions &gt; 3
| sort - sessions</query>
        </search>

      <title>RDP Inbound from External IP - Potential Unauthorized Access</title>
      <table>
        <search>
          <query>index=zeek source="*conn.log"
  id.resp_p=3389
    conn_state IN (SF, S1, S2, S3, OTH, RSTR)
| rename "id.orig_h" as src_ip
         "id.resp_h" as dest_ip
         "id.orig_p" as src_port
         "id.resp_p" as dest_port

| stats count
        min(_time) as firstTime
        max(_time) as lastTime
  BY src_ip dest_ip src_port dest_port proto service

| eval firstTime=strftime(firstTime, "%Y-%m-%dT%H:%M:%S"),
       lastTime=strftime(lastTime, "%Y-%m-%dT%H:%M:%S")

| where NOT (
    src_ip IN ("10.1.2.3", "10.4.5.6")      
    OR dest_ip IN ("10.10.1.5", "10.10.1.6") 
  )</query>
          <earliest>$time_range.earliest$</earliest>
          <latest>$time_range.latest$</latest>
          <sampleRatio>1</sampleRatio>
